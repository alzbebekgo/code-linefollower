// ---------------- PARAMETER (EEPROM) ----------------
struct Param {
  int kp;        // x0.001
  int kd;        // x0.01
  int baseSpd;
  int turnSpd;
  int minSpd;
  int maxSpd;
  int anim;      // 1 = animasi wajah aktif saat jalan
  byte magic;
};
Param P;

const char* names[N_ITEMS] = {"MULAI", "KALIBRASI", "TES SENSOR", "ANIMASI", "Kp", "Kd",
                              "Kec dasar", "Kec belok", "Kec min", "Kec max", "Anim jalan"};

int* itemPtr(byte i) {
  switch (i) {
    case 4: return &P.kp;
    case 5: return &P.kd;
    case 6: return &P.baseSpd;
    case 7: return &P.turnSpd;
    case 8: return &P.minSpd;
    case 9: return &P.maxSpd;
    default: return &P.anim;
  }
}

// ---------------- TOMBOL ----------------
// A6 di-pull-up 10k ke 5V. OK = langsung GND, UP = 1k ke GND, DOWN = 4.7k ke GND
byte readBtn() {
  int a = analogRead(BTN_PIN);
  if (a > 900) return 0;
  if (a < 40)  return 3;   // OK
  if (a < 200) return 1;   // UP
  if (a < 600) return 2;   // DOWN
  return 0;
}

byte getBtn() {
  byte b = readBtn();
  if (!b) return 0;
  delay(25);
  if (readBtn() != b) return 0;
  unsigned long t = millis();
  while (readBtn() && millis() - t < 350) ;   // tahan lama = auto-repeat
  return b;
}

// ---------------- OLED: PESAN & MENU ----------------
void showMsg(const char* a, const char* b, const char* c) {
  oled.firstPage();
  do {
    oled.setFont(u8g2_font_6x10_tr);
    oled.drawStr(0, 14, a);
    oled.drawStr(0, 34, b);
    oled.drawStr(0, 48, c);
  } while (oled.nextPage());
}

void drawMenu(byte cur, bool edit) {
  byte top = (cur > 2) ? cur - 2 : 0;
  if (top > N_ITEMS - 6) top = N_ITEMS - 6;
  oled.firstPage();
  do {
    oled.setFont(u8g2_font_6x10_tr);
    for (byte r = 0; r < 6; r++) {
      byte i = top + r;
      if (i >= N_ITEMS) break;
      int y = (r + 1) * 10 + 2;
      oled.setCursor(0, y);
      oled.print(i == cur ? (edit ? '*' : '>') : ' ');
      oled.print(' ');
      oled.print(names[i]);
      if (i >= 4) {
        oled.setCursor(84, y);
        if (i == 4) oled.print(P.kp * 0.001, 3);
        else if (i == 5) oled.print(P.kd * 0.01, 2);
        else if (i == 10) oled.print(P.anim ? "ON" : "OFF");
        else oled.print(*itemPtr(i));
      }
    }
  } while (oled.nextPage());
}

// gelombang segitiga -a..+a dengan periode p (ms): gerakan kontinu, tidak patah
int tri(unsigned long p, int a) {
  long ph = aTime % p;
  long h = p / 2;
  long x = (ph < h) ? ph : (p - ph);
  return (int)(x * 2 * a / h - a);
}

// 0..range-1 berulang tiap p ms
int saw(unsigned long p, int range) {
  return (int)((aTime % p) * range / p);
}

// garis horizontal aman di tepi layar
void hl(int x, int y, int len) {
  if (x < 0) { len += x; x = 0; }
  if (x + len > 128) len = 128 - x;
  if (len > 0) oled.drawHLine(x, y, len);
}

void drawHeart(int x, int y, byte r) {
  oled.drawDisc(x - r, y, r);
  oled.drawDisc(x + r, y, r);
  oled.drawTriangle(x - 2 * r, y + 1, x + 2 * r, y + 1, x, y + 2 * r + 2);
}

void drop(int x, int y) {
  oled.drawDisc(x, y, 2);
  oled.drawTriangle(x - 2, y, x + 2, y, x, y - 5);
}

void sparkle(int x, int y, int r) {
  oled.drawLine(x - r, y, x + r, y);
  oled.drawLine(x, y - r, x, y + r);
}

void smileArc(int x, int y, int r) {
  oled.drawCircle(x, y, r, ARC_DN);
  oled.drawCircle(x, y, r - 1, ARC_DN);
}

      oled.setDrawColor(0);

      oled.drawDisc(78, 50, 2);
      oled.setDrawColor(1);
      break;
    }

    default: {                                       // 0 normal, 3 siap-siap
      int top = 28 - aEyeH / 2;
      byte rad = (aEyeH < 10) ? 1 : 8;
      oled.drawRBox(c1 - 13, top, 26, aEyeH, rad);
      oled.drawRBox(c2 - 13, top, 26, aEyeH, rad);
      if (aMood == 3) {
        oled.drawDisc(64, 52, 4);                    // mulut "o"
      } else {
        smileArc(64, 48, 8);
      }
      break;
    }
  }
}
